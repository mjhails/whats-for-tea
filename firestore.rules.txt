// Paste ALL of this into: Firebase console > kcal-tracker project > Firestore Database > Rules > Publish
//
// It's the existing Kcal Tracker rules plus three new What's for Tea? rules at the bottom.
// Both apps share one Firebase project, so the rules have to cover both.
//
// - Kcal Tracker: each person's own data stays private; the shared food library is open to signed-in users.
// - What's for Tea?: the week plan, meal library and meal photos are shared between everyone signed in.
// - Nobody who isn't signed in can read or write anything.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ---- Kcal Tracker (unchanged) ----
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;

      match /days/{date} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
      }
    }

    match /shared/library {
      allow read, write: if request.auth != null;
    }

    // ---- What's for Tea? ----
    match /tea/{docId} {
      allow read, write: if request.auth != null && docId in ['week', 'library'];
    }

    match /teaImages/{imageId} {
      allow read: if request.auth != null;
      // photos are small JPEG data URLs; keep each well under Firestore's 1 MB limit
      allow create, update: if request.auth != null
        && request.resource.data.d is string
        && request.resource.data.d.size() < 700000;
      allow delete: if request.auth != null;
    }
  }
}
